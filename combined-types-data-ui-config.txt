export interface Category {
  id: string;
  name: string;
  slug: string;
}

export interface ProductRating {
  average: number; // 0 - 5
  count: number;
}

export interface ProductDimensions {
  length: number; // cm
  width: number; // cm
  height: number; // cm
}

export interface Product {
  // --- Trường gốc (giữ nguyên, không đổi để không phá vỡ UI hiện có) ---
  id: string;
  name: string;
  slug: string;
  price: number;
  categoryId: string;
  thumbUrl: string;
  detailUrl: string;
  description: string;

  // --- Trường mở rộng (thêm mới, đều optional để tương thích ngược) ---
  brand?: string;
  sku?: string;
  originalPrice?: number; // giá gốc trước giảm
  discountPercent?: number; // % giảm giá
  unit?: string; // đơn vị tính: cái, bộ, kg, chai...
  shortDescription?: string; // mô tả ngắn dùng ở trang danh mục
  gallery?: string[]; // ảnh phụ ngoài thumbUrl/detailUrl
  specifications?: Record<string, string>; // thông số kỹ thuật, tuỳ theo ngành hàng
  rating?: ProductRating;
  stock?: number; // tồn kho
  soldCount?: number; // đã bán
  weight?: number; // gram
  dimensions?: ProductDimensions;
  origin?: string; // xuất xứ
  warranty?: string; // chính sách bảo hành/đổi trả
  tags?: string[];
  createdAt?: string; // ISO date string
}

export interface CartItem {
  productId: string;
  quantity: number;
  addedAt: number;
}

import type { CartItem, Product } from "@repo/types";
import { ArrowLeft, Minus, Plus, ShoppingCart, Trash2 } from "lucide-react";
import { Button } from "./Button";

interface CartPageViewProps {
  items: CartItem[];
  products: Product[];
  total: number;
  onUpdateQuantity: (productId: string, quantity: number) => void;
  onRemove: (productId: string) => void;
}

export function CartPageView({
  items,
  products,
  total,
  onUpdateQuantity,
  onRemove,
}: CartPageViewProps) {
  const productMap = new Map(products.map((p) => [p.id, p]));
  const itemCount = items.reduce((sum, i) => sum + i.quantity, 0);

  if (items.length === 0) {
    return (
      <div className="mx-auto flex max-w-2xl flex-col items-center gap-4 px-4 py-24 text-center">
        <div className="flex h-20 w-20 items-center justify-center rounded-full bg-surface">
          <ShoppingCart className="h-9 w-9 text-muted" strokeWidth={1.5} />
        </div>
        <h1 className="text-xl font-semibold text-foreground">
          Giỏ hàng của bạn đang trống
        </h1>
        <p className="text-sm text-muted">
          Hãy khám phá sản phẩm và thêm vào giỏ hàng để bắt đầu mua sắm.
        </p>
        <a href="/">
          <Button variant="primary">Tiếp tục mua sắm</Button>
        </a>
      </div>
    );
  }

  return (
    <div className="mx-auto max-w-6xl px-4 py-8">
      <div className="mb-6 flex items-center justify-between">
        <h1 className="text-2xl font-semibold text-foreground">
          Giỏ hàng
          <span className="ml-2 align-middle text-base font-normal text-muted">
            ({itemCount} sản phẩm)
          </span>
        </h1>
        <a
          href="/"
          className="hidden items-center gap-1 text-sm text-accent hover:text-accent-dark sm:flex"
        >
          <ArrowLeft className="h-4 w-4" />
          Tiếp tục mua sắm
        </a>
      </div>

      <div className="grid grid-cols-1 gap-8 lg:grid-cols-3">
        <div className="flex flex-col gap-4 lg:col-span-2">
          {items.map((item) => {
            const product = productMap.get(item.productId);
            if (!product) return null;
            const lineTotal = product.price * item.quantity;

            return (
              <div
                key={item.productId}
                className="flex gap-4 rounded-card border border-border bg-background p-4 shadow-sm"
              >
                <img
                  src={product.thumbUrl}
                  alt={product.name}
                  width={96}
                  height={96}
                  className="h-24 w-24 flex-shrink-0 rounded-lg bg-surface object-cover"
                />

                <div className="flex flex-1 flex-col justify-between gap-3">
                  <div className="flex items-start justify-between gap-2">
                    <div>
                      <p className="font-medium text-foreground line-clamp-2">
                        {product.name}
                      </p>
                      <p className="mt-1 text-sm text-muted">
                        {product.price.toLocaleString("vi-VN")}₫ / sản phẩm
                      </p>
                    </div>
                    <button
                      onClick={() => onRemove(item.productId)}
                      aria-label={`Xoá ${product.name} khỏi giỏ hàng`}
                      className="shrink-0 rounded-md p-1.5 text-muted transition-colors hover:bg-surface hover:text-red-600"
                    >
                      <Trash2 className="h-4 w-4" />
                    </button>
                  </div>

                  <div className="flex items-center justify-between">
                    <div className="flex items-center rounded-md border border-border">
                      <button
                        onClick={() =>
                          onUpdateQuantity(item.productId, item.quantity - 1)
                        }
                        aria-label="Giảm số lượng"
                        className="flex h-8 w-8 items-center justify-center text-foreground transition-colors hover:bg-surface"
                      >
                        <Minus className="h-3.5 w-3.5" />
                      </button>
                      <span className="w-8 text-center text-sm font-medium text-foreground">
                        {item.quantity}
                      </span>
                      <button
                        onClick={() =>
                          onUpdateQuantity(item.productId, item.quantity + 1)
                        }
                        aria-label="Tăng số lượng"
                        className="flex h-8 w-8 items-center justify-center text-foreground transition-colors hover:bg-surface"
                      >
                        <Plus className="h-3.5 w-3.5" />
                      </button>
                    </div>
                    <p className="font-semibold text-foreground">
                      {lineTotal.toLocaleString("vi-VN")}₫
                    </p>
                  </div>
                </div>
              </div>
            );
          })}
        </div>

        <div className="lg:col-span-1">
          <div className="sticky top-24 rounded-card border border-border bg-background p-6 shadow-sm">
            <h2 className="mb-4 font-semibold text-foreground">
              Tóm tắt đơn hàng
            </h2>
            <div className="flex justify-between text-sm text-muted">
              <span>Tạm tính ({itemCount} sản phẩm)</span>
              <span className="text-foreground">
                {total.toLocaleString("vi-VN")}₫
              </span>
            </div>
            <div className="my-4 border-t border-border" />
            <div className="flex justify-between text-base font-semibold text-foreground">
              <span>Tổng cộng</span>
              <span className="text-accent">
                {total.toLocaleString("vi-VN")}₫
              </span>
            </div>
            <Button variant="primary" className="mt-6 w-full" disabled>
              Tiến hành thanh toán
            </Button>
            <p className="mt-2 text-center text-xs text-muted">
              Bản mô phỏng — không xử lý thanh toán thực tế.
            </p>
          </div>
        </div>
      </div>
    </div>
  );
}

import type { ComponentType, ReactNode } from "react";
import { ShoppingCart } from "lucide-react";

  /** "T&Q Mart (CSR)" hoặc "T&Q Mart (SSR)" — chỉ khác nhau ở nhãn để phân biệt lúc test, không phải khác biệt nghiệp vụ */

  cartQuantity: number;
  /**
   * Mỗi app dùng 1 hệ thống routing khác nhau (react-router-dom vs next/link),
   * nên Link được "tiêm" từ ngoài vào — cùng 1 cách làm với `cartAction` trong
   * ProductDetailView — để phần markup/CSS ở đây luôn render giống hệt nhau
   * giữa CSR và SSR.
   */
  LinkComponent: ComponentType<{ href: string; className?: string; children: ReactNode }>;
}

export function SiteHeader({ storeLabel, cartQuantity, LinkComponent }: SiteHeaderProps) {
  return (
    <header className="flex items-center justify-between bg-background shadow-sm rounded-card px-6 py-3 mt-4 mb-6">
      <LinkComponent href="/" className="flex items-center gap-2 font-semibold text-foreground">
        <span className="w-6 h-6 rounded-md bg-accent text-white text-xs flex items-center justify-center">
          T
        </span>
        {storeLabel}
      </LinkComponent>
      <LinkComponent
        href="/cart"
        className="flex items-center gap-1.5 text-sm text-accent hover:text-accent-dark"
      >
        <ShoppingCart className="w-4 h-4" aria-hidden="true" />
        <span>{cartQuantity}</span>
      </LinkComponent>
    </header>
  );
}

import type { CartItem, Product } from "@repo/types";

interface CartDrawerProps {
  items: CartItem[];
  products: Product[];
  total: number;
  onUpdateQuantity: (productId: string, quantity: number) => void;
  onRemove: (productId: string) => void;
  onClose: () => void;
}

export function CartDrawer({
  items,
  products,
  total,
  onUpdateQuantity,
  onRemove,
  onClose,
}: CartDrawerProps) {
  const productMap = new Map(products.map((p) => [p.id, p]));

  return (
    <div className="fixed inset-y-0 right-0 w-full sm:w-96 bg-background shadow-xl flex flex-col z-50">
      <div className="flex items-center justify-between p-4 border-b border-border">
        <h2 className="font-semibold text-foreground">Giỏ hàng</h2>
        <button onClick={onClose} aria-label="Đóng" className="text-muted hover:text-foreground">✕</button>
      </div>

      <div className="flex-1 overflow-y-auto p-4 flex flex-col gap-4">
        {items.length === 0 && <p className="text-sm text-muted">Giỏ hàng trống.</p>}
        {items.map((item) => {
          const product = productMap.get(item.productId);
          if (!product) return null;
          return (
            <div key={item.productId} className="flex gap-3">
              <img
                src={product.thumbUrl}
                alt={product.name}
                width={64}
                height={64}
                className="w-16 h-16 object-cover rounded bg-surface"
              />
              <div className="flex-1">
                <p className="text-sm font-medium line-clamp-1 text-foreground">{product.name}</p>
                <p className="text-sm text-muted">{product.price.toLocaleString("vi-VN")}₫</p>
                <div className="flex items-center gap-2 mt-1">
                  <button
                    onClick={() => onUpdateQuantity(item.productId, item.quantity - 1)}
                    className="w-6 h-6 border border-border rounded text-foreground hover:bg-surface"
                  >
                    −
                  </button>
                  <span className="text-sm w-6 text-center text-foreground">{item.quantity}</span>
                  <button
                    onClick={() => onUpdateQuantity(item.productId, item.quantity + 1)}
                    className="w-6 h-6 border border-border rounded text-foreground hover:bg-surface"
                  >
                    +
                  </button>
                  <button
                    onClick={() => onRemove(item.productId)}
                    className="text-xs text-red-600 ml-2 hover:text-red-700"
                  >
                    Xoá
                  </button>
                </div>
              </div>
            </div>
          );
        })}
      </div>

      <div className="p-4 border-t border-border">
        <div className="flex justify-between font-semibold text-foreground">
          <span>Tổng tiền</span>
          <span>{total.toLocaleString("vi-VN")}₫</span>
        </div>
      </div>
    </div>
  );
}
