import { create } from "zustand";
import { persist } from "zustand/middleware";
import type { CartItem, Product } from "@repo/types";

interface CartState {
  items: CartItem[];
  addItem: (productId: string, quantity?: number) => void;
  updateQuantity: (productId: string, quantity: number) => void;
  removeItem: (productId: string) => void;
  clearCart: () => void;
  getTotalQuantity: () => number;
}

export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],

      addItem: (productId, quantity = 1) => {
        set((state) => {
          const existing = state.items.find((i) => i.productId === productId);
          if (existing) {
            return {
              items: state.items.map((i) =>
                i.productId === productId
                  ? { ...i, quantity: i.quantity + quantity }
                  : i
              ),
            };
          }
          return {
            items: [...state.items, { productId, quantity, addedAt: Date.now() }],
          };
        });
      },

      updateQuantity: (productId, quantity) => {
        set((state) => ({
          items:
            quantity <= 0
              ? state.items.filter((i) => i.productId !== productId)
              : state.items.map((i) =>
                  i.productId === productId ? { ...i, quantity } : i
                ),
        }));
      },

      removeItem: (productId) => {
        set((state) => ({
          items: state.items.filter((i) => i.productId !== productId),
        }));
      },

      clearCart: () => set({ items: [] }),

      getTotalQuantity: () =>
        get().items.reduce((sum, i) => sum + i.quantity, 0),
    }),
    { name: "tqmart-cart" }
  )
);

export function selectCartTotal(items: CartItem[], products: Product[]): number {
  const priceMap = new Map(products.map((p) => [p.id, p.price]));
  return items.reduce((sum, item) => {
    const price = priceMap.get(item.productId) ?? 0;
    return sum + price * item.quantity;
  }, 0);
}

import type { ButtonHTMLAttributes, ReactNode } from "react";

interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  children: ReactNode;
  variant?: "primary" | "secondary";
}

export function Button({ children, variant = "primary", className = "", ...props }: ButtonProps) {
  const base =
    "inline-flex items-center justify-center gap-2 px-4 py-2 rounded-md font-medium transition-colors disabled:opacity-50 disabled:cursor-not-allowed disabled:pointer-events-none";
  const variants = {
    primary: "bg-accent text-white hover:bg-accent-dark",
    secondary: "bg-surface text-foreground border border-border hover:bg-border",
  };
  return (
    <button className={`${base} ${variants[variant]} ${className}`} {...props}>
      {children}
    </button>
  );
}

import type { CartItem, Product } from "@repo/types";
import { ArrowLeft, Minus, Plus, ShoppingCart, Trash2 } from "lucide-react";
import { Button } from "./Button";
import type { LinkComponentType } from "./types";

interface CartPageViewProps {
  items: CartItem[];
  products: Product[];
  total: number;
  LinkComponent: LinkComponentType;
  onUpdateQuantity: (productId: string, quantity: number) => void;
  onRemove: (productId: string) => void;
}

export function CartPageView({
  items,
  products,
  total,
  LinkComponent,
  onUpdateQuantity,
  onRemove,
}: CartPageViewProps) {
  const productMap = new Map(products.map((p) => [p.id, p]));
  const itemCount = items.reduce((sum, i) => sum + i.quantity, 0);

  if (items.length === 0) {
    return (
      <div className="mx-auto flex max-w-2xl flex-col items-center gap-4 px-4 py-24 text-center">
        <div className="flex h-20 w-20 items-center justify-center rounded-full bg-surface">
          <ShoppingCart className="h-9 w-9 text-muted" strokeWidth={1.5} />
        </div>
        <h1 className="text-xl font-semibold text-foreground">
          Giỏ hàng của bạn đang trống
        </h1>
        <p className="text-sm text-muted">
          Hãy khám phá sản phẩm và thêm vào giỏ hàng để bắt đầu mua sắm.
        </p>
        <LinkComponent href="/">
          <Button variant="primary">Tiếp tục mua sắm</Button>
        </LinkComponent>
      </div>
    );
  }

  return (
    <div className="mx-auto max-w-6xl px-4 py-8">
      <div className="mb-6 flex items-center justify-between">
        <h1 className="text-2xl font-semibold text-foreground">
          Giỏ hàng
          <span className="ml-2 align-middle text-base font-normal text-muted">
            ({itemCount} sản phẩm)
          </span>
        </h1>

        <LinkComponent
          href="/"
          className="hidden items-center gap-1 text-sm text-accent hover:text-accent-dark sm:flex"
        >
          <ArrowLeft className="h-4 w-4" />
          Tiếp tục mua sắm
        </LinkComponent>
      </div>

      <div className="grid grid-cols-1 gap-8 lg:grid-cols-3">
        <div className="flex flex-col gap-4 lg:col-span-2">
          {items.map((item) => {
            const product = productMap.get(item.productId);
            if (!product) return null;
            const lineTotal = product.price * item.quantity;

            return (
              <div
                key={item.productId}
                className="flex gap-4 rounded-card border border-border bg-background p-4 shadow-sm"
              >
                <img
                  src={product.thumbUrl}
                  alt={product.name}
                  width={96}
                  height={96}
                  className="h-24 w-24 flex-shrink-0 rounded-lg bg-surface object-cover"
                />

                <div className="flex flex-1 flex-col justify-between gap-3">
                  <div className="flex items-start justify-between gap-2">
                    <div>
                      <p className="font-medium text-foreground line-clamp-2">
                        {product.name}
                      </p>
                      <p className="mt-1 text-sm text-muted">
                        {product.price.toLocaleString("vi-VN")}₫ / sản phẩm
                      </p>
                    </div>
                    <button
                      onClick={() => onRemove(item.productId)}
                      aria-label={`Xoá ${product.name} khỏi giỏ hàng`}
                      className="shrink-0 rounded-md p-1.5 text-muted transition-colors hover:bg-surface hover:text-red-600"
                    >
                      <Trash2 className="h-4 w-4" />
                    </button>
                  </div>

                  <div className="flex items-center justify-between">
                    <div className="flex items-center rounded-md border border-border">
                      <button
                        onClick={() =>
                          onUpdateQuantity(item.productId, item.quantity - 1)
                        }
                        aria-label="Giảm số lượng"
                        className="flex h-8 w-8 items-center justify-center text-foreground transition-colors hover:bg-surface"
                      >
                        <Minus className="h-3.5 w-3.5" />
                      </button>
                      <span className="w-8 text-center text-sm font-medium text-foreground">
                        {item.quantity}
                      </span>
                      <button
                        onClick={() =>
                          onUpdateQuantity(item.productId, item.quantity + 1)
                        }
                        aria-label="Tăng số lượng"
                        className="flex h-8 w-8 items-center justify-center text-foreground"
                      >
                        <Plus className="h-3.5 w-3.5" />
                      </button>
                    </div>
                    <p className="font-semibold text-foreground">
                      {lineTotal.toLocaleString("vi-VN")}₫
                    </p>
                  </div>
                </div>
              </div>
            );
          })}
        </div>

        <div className="lg:col-span-1">
          <div className="sticky top-24 rounded-card border border-border bg-background p-6 shadow-sm">
            <h2 className="mb-4 font-semibold text-foreground">
              Tóm tắt đơn hàng
            </h2>
            <div className="flex justify-between text-sm text-muted">
              <span>Tạm tính ({itemCount} sản phẩm)</span>
              <span className="text-foreground">
                {total.toLocaleString("vi-VN")}₫
              </span>
            </div>
            <div className="my-4 border-t border-border" />
            <div className="flex justify-between text-base font-semibold text-foreground">
              <span>Tổng cộng</span>
              <span className="text-accent">
                {total.toLocaleString("vi-VN")}₫
              </span>
            </div>
            <Button variant="primary" className="mt-6 w-full" disabled>
              Tiến hành thanh toán
            </Button>
            <p className="mt-2 text-center text-xs text-muted">
              Bản mô phỏng — không xử lý thanh toán thực tế.
            </p>
          </div>
        </div>
      </div>
    </div>
  );
}

// 📄 FILE: apps/app-csr/src/App.tsx
// ======================================================================

import { Routes, Route } from "react-router-dom";
import { ProductListPage } from "./pages/ProductListPage";
import { ProductDetailPage } from "./pages/ProductDetailPage";
import { CartPage } from "./pages/CartPage";
import { useCartStore } from "@repo/cart-store";
import { SiteHeader } from "@repo/ui";
import { RouterLink } from "./components/RouterLink";

function App() {
  const totalQuantity = useCartStore((s) => s.getTotalQuantity());

  return (
    <div className="min-h-screen bg-surface">
      <div className="max-w-6xl mx-auto px-4">
        <SiteHeader
          storeLabel="T&Q Mart (CSR)"
          cartQuantity={totalQuantity}
          LinkComponent={RouterLink}
        />
        <Routes>
          <Route path="/" element={<ProductListPage />} />
          <Route path="/products/:id" element={<ProductDetailPage />} />
          <Route path="/cart" element={<CartPage />} />
        </Routes>
      </div>
    </div>
  );
}

export default App;

import type { ReactNode } from "react";
import { Link } from "react-router-dom";

export function RouterLink({
  href,
  className,
  children,
}: {
  href: string;
  className?: string;
  children: ReactNode;
}) {
  return (
    <Link to={href} className={className}>
      {children}
    </Link>
  );
}

import { ProductCard, AddToCartButton } from "@repo/ui";
import { products } from "@repo/data";
import { Link } from "react-router-dom";

export function ProductListPage() {
  return (
    <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-5 md:gap-6 pb-8">

// ======================================================================
// 📄 FILE: apps/app-csr/src/pages/ProductDetailPage.tsx
// ======================================================================

import { useParams } from "react-router-dom";
import { ProductDetailView, AddToCartButton } from "@repo/ui";
import { products } from "@repo/data";

export function ProductDetailPage() {
  const { id } = useParams();
  const product = products.find((p) => p.id === id);

  if (!product) return <div className="p-4">Không tìm thấy sản phẩm.</div>;

  return (
    <ProductDetailView
      product={product}
      cartAction={<AddToCartButton productId={product.id} />}
    />
  );
}

// ======================================================================
// 📄 FILE: apps/app-csr/src/pages/CartPage.tsx
// ======================================================================

import { CartPageView } from "@repo/ui";
import { useCartStore, selectCartTotal } from "@repo/cart-store";
import { products } from "@repo/data";
import { RouterLink } from "../components/RouterLink";

export function CartPage() {
  const items = useCartStore((s) => s.items);
  const updateQuantity = useCartStore((s) => s.updateQuantity);
  const removeItem = useCartStore((s) => s.removeItem);
  const total = selectCartTotal(items, products);

  return (
    <CartPageView
      items={items}
      products={products}
      total={total}
      onUpdateQuantity={updateQuantity}
      onRemove={removeItem}
      LinkComponent={RouterLink}
    />
  );
}

"use client";

import { CartPageView } from "@repo/ui";
import { useCartStore, selectCartTotal } from "@repo/cart-store";
import { products } from "@repo/data";
import { useEffect, useState } from "react";
import { NextLinkAdapter } from "../next-link-adapter";

export default function CartPage() {
  const items = useCartStore((s) => s.items);
  const updateQuantity = useCartStore((s) => s.updateQuantity);
  const removeItem = useCartStore((s) => s.removeItem);
  const [mounted, setMounted] = useState(false);

  useEffect(() => setMounted(true), []);

  const total = selectCartTotal(items, products);

  if (!mounted) return <div className="p-4">Đang tải giỏ hàng...</div>;

  return (
    <CartPageView
      items={items}
      products={products}
      total={total}
      onUpdateQuantity={updateQuantity}
      onRemove={removeItem}
      LinkComponent={NextLinkAdapter}
    />
  );
}
